14/// Window level constants for NSPanel
15/// Based on NSWindow.Level constants from macOS
16#[derive(Debug, Clone, Copy, PartialEq)]
17pub enum PanelLevel {
18    /// Normal window level (0)
19    Normal,
20    /// Submenu window level (3)
21    Submenu,
22    /// Torn-off menu window level (3)
23    TornOffMenu,
24    /// Floating window level (4)
25    Floating,
26    /// Modal panel window level (8)
27    ModalPanel,
28    /// Utility window level (19)
29    Utility,
30    /// Dock window level (20)
31    Dock,
32    /// Main menu window level (24)
33    MainMenu,
34    /// Status window level (25)
35    Status,
36    /// Pop-up menu window level (101)
37    PopUpMenu,
38    /// Screen saver window level (1000)
39    ScreenSaver,
40    /// Custom level value
41    Custom(i32),
42}
43
44impl PanelLevel {
45    /// Convert to the raw i64 value used by NSWindow
46    pub fn value(&self) -> i64 {
47        match self {
48            PanelLevel::Normal => 0,
49            PanelLevel::Submenu => 3,
50            PanelLevel::TornOffMenu => 3,
51            PanelLevel::Floating => 4,
52            PanelLevel::ModalPanel => 8,
53            PanelLevel::Utility => 19,
54            PanelLevel::Dock => 20,
55            PanelLevel::MainMenu => 24,
56            PanelLevel::Status => 25,
57            PanelLevel::PopUpMenu => 101,
58            PanelLevel::ScreenSaver => 1000,
59            PanelLevel::Custom(value) => *value as i64,
60        }
61    }
62}
63
64impl From<PanelLevel> for i64 {
65    fn from(level: PanelLevel) -> Self {
66        level.value()
67    }
68}
69
70impl From<i32> for PanelLevel {
71    fn from(value: i32) -> Self {
72        PanelLevel::Custom(value)
73    }
74}
75
76impl From<i64> for PanelLevel {
77    fn from(value: i64) -> Self {
78        PanelLevel::Custom(value as i32)
79    }

479/// Builder for creating panels with Tauri-like API
480///
481/// The builder provides a fluent interface for creating NSPanel windows with
482/// various configurations. It can work with both the default panel type or
483/// custom panel classes created with the `panel!` macro.
484///
485/// # Type Parameters
486/// - `R`: The Tauri runtime type
487/// - `T`: The panel type (must implement `FromWindow<R>`)
488///
489/// # Example
490/// ```rust
491/// use tauri_nspanel::{panel, PanelBuilder, PanelLevel};
492///
493/// // Using default panel type
494/// let panel = PanelBuilder::new(&app, "my-panel")
495///     .url(WebviewUrl::App("panel.html".into()))
496///     .title("Tool Panel")
497///     .level(PanelLevel::Floating)
498///     .build()?;
499///
500/// // Using custom panel type
501/// panel!(CustomPanel {
502///     config: {
503///         canBecomeKeyWindow: false
504///     }
505/// });
506///
507/// let custom = PanelBuilder::<_, CustomPanel>::new(&app, "custom")
508///     .url(WebviewUrl::App("custom.html".into()))
509///     .build()?;
510/// ```
511pub struct PanelBuilder<'a, R: Runtime, T: FromWindow<R> + 'static> {
512    handle: &'a AppHandle<R>,
513    label: String,
514    url: Option<WebviewUrl>,
515    title: Option<String>,
516    position: Option<Position>,
517    size: Option<Size>,
518    pub(crate) panel_config: PanelConfig,
519    window_fn: Option<WindowConfigFn<'a, R>>,
520    _phantom: std::marker::PhantomData<T>,
521}

704    /// Apply a custom configuration function to the WebviewWindowBuilder

727        F: FnOnce(

743        );

745        if let Some(title) = self.title {
746            window_builder = window_builder.title(title);
747        }
748
749        if let Some(position) = self.position {
750            match position {
751                Position::Physical(pos) => {
752                    window_builder = window_builder.position(pos.x as f64, pos.y as f64);
753                }
754                Position::Logical(pos) => {
755                    window_builder = window_builder.position(pos.x, pos.y);
756                }
757            }
758        }
759
760        if let Some(size) = self.size {
761            match size {
762                Size::Physical(s) => {
763                    window_builder = window_builder.inner_size(s.width as f64, s.height as f64);
764                }
765                Size::Logical(s) => {
766                    window_builder = window_builder.inner_size(s.width, s.height);
767                }
768            }
769        }
770
771        // Apply custom configuration if provided
772        if let Some(window_fn) = self.window_fn {
773            window_builder = window_fn(window_builder);
774        }
775
776        // Build the window
777        let window = window_builder.build()?;
778
779        // Convert to panel
780        let panel = window.to_panel::<T>().unwrap();
781
782        // Apply panel configuration using the Panel trait methods
783        if let Some(floating) = self.panel_config.floating {
784            panel.set_floating_panel(floating);
785        }
786        if let Some(level) = self.panel_config.level {
787            panel.set_level(level.value());
788        }
789        if let Some(has_shadow) = self.panel_config.has_shadow {
790            panel.set_has_shadow(has_shadow);
791        }
792        if let Some(opaque) = self.panel_config.opaque {
793            panel.set_opaque(opaque);
794        }
795        if let Some(alpha_value) = self.panel_config.alpha_value {
796            panel.set_alpha_value(alpha_value);
797        }
798        if let Some(hides) = self.panel_config.hides_on_deactivate {
799            panel.set_hides_on_deactivate(hides);
800        }
801        if let Some(value) = self.panel_config.becomes_key_only_if_needed {
802            panel.set_becomes_key_only_if_needed(value);
803        }
804        if let Some(value) = self.panel_config.accepts_mouse_moved_events {
805            panel.set_accepts_mouse_moved_events(value);
806        }
807        if let Some(value) = self.panel_config.ignores_mouse_events {
808            panel.set_ignores_mouse_events(value);
809        }
810        if let Some(value) = self.panel_config.movable_by_window_background {
811            panel.set_movable_by_window_background(value);
812        }
813        if let Some(value) = self.panel_config.released_when_closed {
814            panel.set_released_when_closed(value);
815        }
816        if let Some(value) = self.panel_config.works_when_modal {
817            panel.set_works_when_modal(value);
818        }
819        if let Some(style_mask) = self.panel_config.style_mask {
820            panel.set_style_mask(style_mask.0);
821        }
822        if let Some(behavior) = self.panel_config.collection_behavior {
823            panel.set_collection_behavior(behavior.0);
824        }
825
826        Ok(panel)
827    }
